async function handleFormSubmit(e, formType) { e.preventDefault(); // 1. Έλεγχος περιοχών const containerId = formType === 'worker' ? 'worker-locations' : 'employer-locations'; const checkedBoxes = document.querySelectorAll(`#${containerId} .com-checkbox:checked`); if (checkedBoxes.length === 0) { alert('Παρακαλώ επιλέξτε τουλάχιστον μία περιοχή ή δήμο.'); return; } // 2. Έλεγχος Ειδικότητας/Εργασίας const specialty = document.getElementById(formType === 'worker' ? 'w_specialty' : 'e_specialty').value; const task = document.getElementById(formType === 'worker' ? 'w_task' : 'e_task').value; if (!specialty && !task) { alert('Παρακαλώ επιλέξτε τουλάχιστον μία Ειδικότητα Ή ένα Είδος Εργασίας.'); return; } // 3. Έλεγχος Επικοινωνίας const phone = document.getElementById(formType === 'worker' ? 'w_phone' : 'e_phone').value.trim(); const email = document.getElementById(formType === 'worker' ? 'w_email' : 'e_email').value.trim(); if (!phone && !email) { alert('Πρέπει να συμπληρώσετε τουλάχιστον ένα τηλέφωνο ή ένα e-mail επικοινωνίας.'); return; } // 4. Μάζεμα όλων των δεδομένων const formData = { name: document.getElementById(formType === 'worker' ? 'w_name' : 'e_name').value.trim(), specialty: specialty, task: task, locations: Array.from(checkedBoxes).map(cb => cb.value), phone: phone, email: email, comments: document.getElementById(formType === 'worker' ? 'w_comments' : 'e_comments').value.trim() }; // 5. Το σωστό URL του Cloudflare Worker που φτιάξαμε const workerUrl = formType === 'worker' ? 'https://jobs.kafevolta.workers.dev/api/worker' : 'https://jobs.kafevolta.workers.dev/api/employer'; // 6. Αποστολή στη βάση try { // Αλλάζουμε το κουμπί για να δείξουμε ότι φορτώνει const btn = e.target.querySelector('button[type="submit"]'); const originalText = btn.innerText; btn.innerText = 'Αποστολή...'; btn.disabled = true; const response = await fetch(workerUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(formData) }); if (response.ok) { alert('Η καταχώρηση ολοκληρώθηκε με επιτυχία!'); e.target.reset(); // Καθαρίζει τα πεδία κειμένου // Καθαρίζει τα τικ από τα checkboxes document.querySelectorAll(`#${containerId} .com-checkbox`).forEach(cb => cb.checked = false); } else { alert('Υπήρξε πρόβλημα κατά την αποθήκευση. Δοκιμάστε ξανά.'); } // Επαναφέρουμε το κουμπί btn.innerText = originalText; btn.disabled = false; } catch (error) { console.error('Σφάλμα:', error); alert('Αδυναμία σύνδεσης. Ελέγξτε το ίντερνετ σας.'); } }
